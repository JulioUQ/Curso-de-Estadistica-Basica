Laboratorio estadístico

Aprende manipulando: en cada experimento cambias los parámetros, el gráfico se actualiza al momento y un texto explica lo que estás viendo. No hace falta instalar nada: todo se calcula en tu navegador. Las simulaciones usan una semilla, así que con los mismos controles obtendrás siempre el mismo resultado y podrás comprobarlo o comentarlo con otras personas.

function Phi(x) {
  const t = 1 / (1 + 0.3275911 * Math.abs(x) / Math.SQRT2);
  const poly = ((((1.061405429 * t - 1.453152027) * t + 1.421413741) * t - 0.284496736) * t + 0.254829592) * t;
  const erf = 1 - poly * Math.exp(-x * x / 2);
  return x >= 0 ? (1 + erf) / 2 : (1 - erf) / 2;
}
function dnorm(x, mu, sigma) {
  const z = (x - mu) / sigma;
  return Math.exp(-0.5 * z * z) / (sigma * Math.sqrt(2 * Math.PI));
}
// ln Γ(x) (Lanczos, g = 7) y beta incompleta regularizada (fracción continua de Lentz)
function lnGamma(x) {
  const c = [0.99999999999980993, 676.5203681218851, -1259.1392167224028, 771.32342877765313,
    -176.61502916214059, 12.507343278686905, -0.13857109526572012, 9.9843695780195716e-6, 1.5056327351493116e-7];
  if (x < 0.5) return Math.log(Math.PI / Math.sin(Math.PI * x)) - lnGamma(1 - x);
  x -= 1;
  let a = c[0];
  const t = x + 7.5;
  for (let i = 1; i < 9; i++) a += c[i] / (x + i);
  return 0.5 * Math.log(2 * Math.PI) + (x + 0.5) * Math.log(t) - t + Math.log(a);
}
function betaReg(x, a, b) {
  if (x <= 0) return 0;
  if (x >= 1) return 1;
  if (x > (a + 1) / (a + b + 2)) return 1 - betaReg(1 - x, b, a);
  const lnFront = a * Math.log(x) + b * Math.log(1 - x) - (lnGamma(a) + lnGamma(b) - lnGamma(a + b));
  let f = 1, C = 1, D = 0;
  for (let i = 0; i <= 300; i++) {
    const m = Math.floor(i / 2);
    let num;
    if (i === 0) num = 1;
    else if (i % 2 === 0) num = (m * (b - m) * x) / ((a + 2 * m - 1) * (a + 2 * m));
    else num = -((a + m) * (a + b + m) * x) / ((a + 2 * m) * (a + 2 * m + 1));
    D = 1 + num * D; if (Math.abs(D) < 1e-30) D = 1e-30; D = 1 / D;
    C = 1 + num / C; if (Math.abs(C) < 1e-30) C = 1e-30;
    const cd = C * D;
    f *= cd;
    if (Math.abs(1 - cd) < 1e-12) break;
  }
  return Math.exp(lnFront) * (f - 1) / a;
}
// Función de distribución y cuantil de la t de Student con df grados de libertad
function tCdf(t, df) {
  const p = 0.5 * betaReg(df / (df + t * t), df / 2, 0.5);
  return t >= 0 ? 1 - p : p;
}
function tQuantile(p, df) {
  if (p === 0.5) return 0;
  if (p < 0.5) return -tQuantile(1 - p, df);
  let lo = 0, hi = 1;
  while (tCdf(hi, df) < p) hi *= 2;
  for (let i = 0; i < 100; i++) {
    const mid = (lo + hi) / 2;
    if (tCdf(mid, df) < p) lo = mid; else hi = mid;
  }
  return (lo + hi) / 2;
}

Para pensar. Fija a y b en 9 y 11 y mueve μ de 0 a 10. ¿Por qué la probabilidad del intervalo cambia tanto, si el intervalo no se mueve?

TipUna posible respuesta

La probabilidad depende de dónde está el intervalo respecto a la distribución. Con μ = 0 y σ = 1, el intervalo (9, 11) está a más de nueve desviaciones típicas de la media y su probabilidad es prácticamente cero; con μ = 10 queda centrado y contiene el 68,3 % (de μ − σ a μ + σ). Lo que importa son los valores tipificados \(z\), no los valores originales.

D. Contrastes de hipótesis y potencia

Objetivo. Ver la región de rechazo de un contraste z sobre una media y cómo el tamaño del efecto, el tamaño muestral y \(\alpha\) determinan los errores de tipo I y II y la potencia.

Contrastamos \(H_0: \mu = \mu_0\) con \(\sigma\) conocida. El tamaño del efecto es \(d = (\mu - \mu_0)/\sigma\), la diferencia real medida en desviaciones típicas. Bajo \(H_0\) el estadístico \(Z = (\bar X - \mu_0)/(\sigma/\sqrt{n})\) sigue una \(N(0, 1)\); si la diferencia real es \(d\), sigue una \(N(d\sqrt{n}, 1)\).
